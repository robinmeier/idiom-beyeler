for N in 9; do
  for T in 3.5; do
    for R in 1 2 3 4; do
      for P in 0.3 0.5; do
        for VOL_LABEL in "norm:0" "norm-6:-6" "norm-12:-12"; do
          LABEL="${VOL_LABEL%%:*}"
          DB="${VOL_LABEL##*:}"

          # Create a temp normalised (+ optional gain) version of the input
          TMPFILE=$(mktemp /tmp/sktrt-input-XXXXXX.wav)
          ffmpeg -y -i "${STABS}/sktrt-${N}.wav" \
            -af "loudnorm,volume=${DB}dB" \
            "$TMPFILE" -loglevel quiet

          uv run clone_voice.py \
            "$TMPFILE" \
            "$TARGET" \
            --repetition-penalty 0.9 \
            --temperature $T \
            --top-p $P \
            -o "${OUT}/sktrt-${N}-t${T}-p${P}-rp0.9-${R}-${LABEL}.wav"

          rm "$TMPFILE"
        done
      done
    done
  done
done

          sox "${STABS}/sktrt-${N}.wav" "$TMPFILE" norm gain $OFFSET

            "$TMPFILE" \
            "$TARGET" \
            --repetition-penalty 0.9 \

            --top-p $P \
            -o "${OUT}/sktrt-${N}-t${T}-p${P}-rp0.9-${R}-${LABEL}.wav"

          rm "$TMPFILE"
        done
      done
    done
  done
done
